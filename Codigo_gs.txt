function doPost(e) {
  var resultado = { status: "error", message: "Solicitud POST inválida" };
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "No se recibieron datos" })).setMimeType(ContentService.MimeType.JSON);
    }
    var data = JSON.parse(e.postData.contents);
    var action = data.action;

    if (action === "login") {
      resultado = procesarLogin_(data.usuario, data.password);
    } else if (action === "crearProducto") {
      resultado = crearOActualizarProducto_(data);
    } else if (action === "editarProducto") {
      resultado = editarProducto_(data);
    } else if (action === "eliminarProducto") {
      resultado = eliminarProducto_(data);
    } else if (action === "eliminarMasivoProductos") {
      resultado = eliminarMasivoProductos_(data);
    } else if (action === "importarMasivoProductos") {
      resultado = importarMasivoProductos_(data);
    } else if (action === "registrarEntrada") {
      resultado = registrarEntradaSalida_("ENTRADAS", data);
    } else if (action === "registrarSalida") {
      resultado = registrarEntradaSalida_("SALIDAS", data);
    } else if (action === "reversarSalida") {
      resultado = reversarSalida_(data);
    } else if (action === "cambiarPassword") {
      resultado = cambiarPassword_(data);
    } else if (action === "crearUsuario") {
      resultado = crearUsuario_(data);
    } else if (action === "editarUsuario") {
      resultado = editarUsuario_(data);
    } else if (action === "eliminarUsuario") {
      resultado = eliminarUsuario_(data);
    } else if (action === "actualizarValorOroDia") {
      resultado = actualizarValorOroDia_(data);
    } else if (action === "crearProveedor") {
      resultado = crearProveedor_(data);
    } else if (action === "editarProveedor") {
      resultado = editarProveedor_(data);
    } else if (action === "eliminarProveedor") {
      resultado = eliminarProveedor_(data);
    }
  } catch (error) {
    resultado = { status: "error", message: "Excepción: " + error.toString() };
  }
  return ContentService.createTextOutput(JSON.stringify(resultado)).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var action = e.parameter.action;
  var resultado = { status: "error", message: "Acción GET no válida" };

  try {
    if (action === "test") {
      resultado = { status: "success", message: "Conexión exitosa con MANU JOYEROS API" };
    } else if (action === "obtenerProductos") {
      resultado = { status: "success", data: obtenerDatosHoja_("PRODUCTOS") };
    } else if (action === "obtenerUsuarios") {
      resultado = { status: "success", data: obtenerDatosHoja_("USUARIOS") };
    } else if (action === "obtenerEntradas") {
      resultado = { status: "success", data: obtenerDatosConDetalleProducto_("ENTRADAS") };
    } else if (action === "obtenerSalidas") {
      resultado = { status: "success", data: obtenerDatosConDetalleProducto_("SALIDAS") };
    } else if (action === "obtenerCategorias") {
      resultado = obtenerCategorias_();
    } else if (action === "obtenerMateriales") {
      resultado = obtenerMateriales_();
    } else if (action === "obtenerColores") {
      resultado = obtenerColores_();
    } else if (action === "obtenerValorOroDia") {
      resultado = obtenerValorOroDia_();
    } else if (action === "obtenerProveedores") {
      resultado = { status: "success", data: obtenerDatosHoja_("PROVEEDORES") };
    }
  } catch (error) {
    resultado = { status: "error", message: error.toString() };
  }

  return ContentService.createTextOutput(JSON.stringify(resultado))
                      .setMimeType(ContentService.MimeType.JSON);
}

function procesarLogin_(usuarioIngresado, passwordIngresado) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hoja = ss.getSheetByName("USUARIOS");
  if (!hoja || hoja.getLastRow() <= 1) return { status: "error", message: "No hay usuarios configurados." };
  
  var rango = hoja.getDataRange().getValues();
  var headers = rango[0];
  var rows = rango.slice(1);
  
  var usuIdx = headers.indexOf("Usuario");
  var passIdx = headers.indexOf("Password_Hash");
  if (passIdx === -1) passIdx = headers.indexOf("Password");
  
  var estadoIdx = headers.indexOf("Estado");
  var nombreIdx = headers.indexOf("Nombre");
  var rolIdx = headers.indexOf("Rol");
  var ultimoAccesoIdx = headers.indexOf("Ultimo_Acceso");
  var fechaVencIdx = headers.indexOf("Fecha_Vencimiento");
  
  for (var i = 0; i < rows.length; i++) {
    var uRow = rows[i];
    var usuarioVal = String(uRow[usuIdx]).trim();
    
    if (usuarioVal.toLowerCase() === String(usuarioIngresado).trim().toLowerCase()) {
      if (String(uRow[passIdx]).trim() !== String(passwordIngresado).trim()) {
        return { status: "error", message: "Usuario o contraseña incorrectos." };
      }
      if (estadoIdx !== -1 && String(uRow[estadoIdx]).toUpperCase() !== "ACTIVO") {
        return { status: "error", message: "El usuario se encuentra inactivo." };
      }
      
      if (fechaVencIdx !== -1 && uRow[fechaVencIdx]) {
        var rawVenc = uRow[fechaVencIdx];
        var fechaVenc = null;

        if (rawVenc instanceof Date) {
          fechaVenc = rawVenc;
        } else {
          var strVenc = String(rawVenc).trim();
          var partes = strVenc.split(/[\/\-]/);
          if (partes.length === 3) {
            if (partes[0].length === 4) {
              fechaVenc = new Date(Number(partes[0]), Number(partes[1]) - 1, Number(partes[2]));
            } else {
              fechaVenc = new Date(Number(partes[2]), Number(partes[1]) - 1, Number(partes[0]));
            }
          } else {
            fechaVenc = new Date(strVenc);
          }
        }

        if (fechaVenc instanceof Date && !isNaN(fechaVenc.getTime())) {
          var hoy = new Date();
          hoy.setHours(0, 0, 0, 0);
          fechaVenc.setHours(0, 0, 0, 0);

          if (fechaVenc.getTime() < hoy.getTime()) {
            var fechaStr = Utilities.formatDate(fechaVenc, "America/Bogota", "yyyy-MM-dd");
            return { status: "error", message: "Acceso denegado: Su cuenta ha vencido (" + fechaStr + ")." };
          }
        }
      }
      
      if (ultimoAccesoIdx !== -1) {
        var fechaActual = Utilities.formatDate(new Date(), "America/Bogota", "yyyy-MM-dd HH:mm:ss");
        var celdaUltimoAcceso = hoja.getRange(i + 2, ultimoAccesoIdx + 1);
        celdaUltimoAcceso.setNumberFormat("@");
        celdaUltimoAcceso.setValue(fechaActual);
      }

      return { 
        status: "success", 
        message: "Login exitoso", 
        usuario: { nombre: uRow[nombreIdx], rol: uRow[rolIdx], usuario: uRow[usuIdx] } 
      };
    }
  }
  return { status: "error", message: "Usuario o contraseña incorrectos." };
}

function cambiarPassword_(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hoja = ss.getSheetByName("USUARIOS");
  var filas = hoja.getDataRange().getValues();
  var headers = filas[0];
  
  var usuIdx = headers.indexOf("Usuario");
  var passIdx = headers.indexOf("Password_Hash");
  if (passIdx === -1) passIdx = headers.indexOf("Password");
  
  if (usuIdx === -1 || passIdx === -1) return { status: "error", message: "Estructura de usuarios no encontrada." };

  for (var r = 1; r < filas.length; r++) {
    if (String(filas[r][usuIdx]).trim().toLowerCase() === String(data.usuario).trim().toLowerCase()) {
      if (String(filas[r][passIdx]).trim() !== String(data.password_actual).trim()) {
        return { status: "error", message: "La contraseña actual es incorrecta." };
      }
      hoja.getRange(r + 1, passIdx + 1).setValue(data.password_nueva);
      registrarEnEspejo_("CAMBIAR_PASSWORD", data.usuario, "Actualización de contraseña de usuario", data.usuario);
      return { status: "success", message: "¡Contraseña actualizada con éxito!" };
    }
  }
  return { status: "error", message: "Usuario no encontrado." };
}

function crearUsuario_(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hoja = ss.getSheetByName("USUARIOS");
  var filas = hoja.getDataRange().getValues();
  var headers = filas[0];
  
  var usuIdx = headers.indexOf("Usuario");
  for (var r = 1; r < filas.length; r++) {
    if (String(filas[r][usuIdx]).trim().toLowerCase() === String(data.usuario).trim().toLowerCase()) {
      return { status: "error", message: "El nombre de usuario ya existe." };
    }
  }

  var nuevaFila = new Array(headers.length);
  headers.forEach(function(h, index) {
    if (h === "Usuario") nuevaFila[index] = data.usuario;
    else if (h === "Password_Hash" || h === "Password") nuevaFila[index] = data.password;
    else if (h === "Nombre") nuevaFila[index] = data.nombre;
    else if (h === "Rol") nuevaFila[index] = data.rol;
    else if (h === "Estado") nuevaFila[index] = data.estado || "ACTIVO";
    else if (h === "Fecha_Vencimiento") nuevaFila[index] = data.fecha_vencimiento;
    else if (h === "Ultimo_Acceso") nuevaFila[index] = "-";
    else nuevaFila[index] = "";
  });

  var filaNum = hoja.getLastRow() + 1;
  hoja.appendRow(nuevaFila);

  var fechaVencIdx = headers.indexOf("Fecha_Vencimiento");
  if (fechaVencIdx !== -1 && data.fecha_vencimiento) {
    var celdaVenc = hoja.getRange(filaNum, fechaVencIdx + 1);
    celdaVenc.setNumberFormat("@");
    celdaVenc.setValue(data.fecha_vencimiento);
  }

  registrarEnEspejo_("CREAR_USUARIO", data.usuario, "Creación de usuario: " + data.nombre + " [Rol: " + data.rol + "]", data.usuario_creador || "ADMIN");
  return { status: "success", message: "¡Usuario creado con éxito!" };
}

function editarUsuario_(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hoja = ss.getSheetByName("USUARIOS");
  var filas = hoja.getDataRange().getValues();
  var headers = filas[0];
  
  var usuIdx = headers.indexOf("Usuario");
  for (var r = 1; r < filas.length; r++) {
    if (String(filas[r][usuIdx]).trim().toLowerCase() === String(data.usuario_original).trim().toLowerCase()) {
      var filaNum = r + 1;
      if (headers.indexOf("Nombre") !== -1 && data.nombre) hoja.getRange(filaNum, headers.indexOf("Nombre") + 1).setValue(data.nombre);
      if (headers.indexOf("Rol") !== -1 && data.rol) hoja.getRange(filaNum, headers.indexOf("Rol") + 1).setValue(data.rol);
      if (headers.indexOf("Estado") !== -1 && data.estado) hoja.getRange(filaNum, headers.indexOf("Estado") + 1).setValue(data.estado);
      
      if (headers.indexOf("Fecha_Vencimiento") !== -1 && data.fecha_vencimiento) {
        var celdaVenc = hoja.getRange(filaNum, headers.indexOf("Fecha_Vencimiento") + 1);
        celdaVenc.setNumberFormat("@");
        celdaVenc.setValue(data.fecha_vencimiento);
      }

      if (headers.indexOf("Password_Hash") !== -1 && data.password) hoja.getRange(filaNum, headers.indexOf("Password_Hash") + 1).setValue(data.password);
      if (headers.indexOf("Password") !== -1 && data.password) hoja.getRange(filaNum, headers.indexOf("Password") + 1).setValue(data.password);
      
      registrarEnEspejo_("EDITAR_USUARIO", data.usuario_original, "Modificación de datos de usuario: " + data.nombre, "ADMIN");
      return { status: "success", message: "¡Usuario actualizado correctamente!" };
    }
  }
  return { status: "error", message: "Usuario no encontrado para editar." };
}

function crearProveedor_(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hoja = ss.getSheetByName("PROVEEDORES");
  if (!hoja) {
    hoja = ss.insertSheet("PROVEEDORES");
    hoja.appendRow(["NIT", "Nombre", "Contacto", "Telefono", "Email", "Direccion", "Especialidad"]);
  }
  
  var filas = hoja.getDataRange().getValues();
  var headers = filas[0];
  var nitIdx = headers.indexOf("NIT");
  
  for (var r = 1; r < filas.length; r++) {
    if (String(filas[r][nitIdx]).trim() === String(data.nit).trim()) {
      return { status: "error", message: "Ya existe un proveedor registrado con ese NIT." };
    }
  }

  hoja.appendRow([
    data.nit,
    data.nombre,
    data.contacto,
    data.telefono,
    data.email,
    data.direccion,
    data.especialidad
  ]);

  registrarEnEspejo_("CREAR_PROVEEDOR", data.nit, "Creación de proveedor: " + data.nombre, data.usuario || "ADMIN");
  return { status: "success", message: "¡Proveedor registrado con éxito!" };
}

function obtenerCategorias_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hoja = ss.getSheetByName("PRODUCTOS");
  var categoriasSet = {};
  
  var bases = ["ANILLOS", "CADENAS", "ARETES", "DIJES", "PULSERAS", "ESTUCHES"];
  bases.forEach(function(b) { categoriasSet[b] = true; });

  if (hoja && hoja.getLastRow() > 1) {
    var valores = hoja.getDataRange().getValues();
    var headers = valores[0];
    var catIdx = headers.indexOf("ID_Categoria");
    
    if (catIdx !== -1) {
      for (var i = 1; i < valores.length; i++) {
        var cat = String(valores[i][catIdx]).trim().toUpperCase();
        if (cat) { categoriasSet[cat] = true; }
      }
    }
  }
  return { status: "success", categorias: Object.keys(categoriasSet).sort() };
}

function obtenerMateriales_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hoja = ss.getSheetByName("PRODUCTOS");
  var materialesSet = {};

  if (hoja && hoja.getLastRow() > 1) {
    var valores = hoja.getDataRange().getValues();
    var headers = valores[0];
    var matIdx = headers.indexOf("Material");
    if (matIdx === -1) matIdx = headers.indexOf("ID_Proveedor");
    
    if (matIdx !== -1) {
      for (var i = 1; i < valores.length; i++) {
        var mat = String(valores[i][matIdx]).trim().toUpperCase();
        if (mat) { materialesSet[mat] = true; }
      }
    }
  }

  if (Object.keys(materialesSet).length === 0) {
    materialesSet["ORO"] = true;
  }

  return { status: "success", materiales: Object.keys(materialesSet).sort() };
}

function obtenerValorOroDia_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var valorOroDia = 250000;
  
  try {
    var hojaConfig = ss.getSheetByName("ORO_DIA") || ss.getSheetByName("CONFIG") || ss.getSheetByName("ACTUALIZACION_ORO");
    if (!hojaConfig) {
      hojaConfig = ss.insertSheet("ORO_DIA");
      hojaConfig.getRange("A1").setValue("VALOR_ORO_DIA");
      hojaConfig.getRange("B1").setValue(250000);
    }
    
    var val = hojaConfig.getRange("B1").getValue();
    if (isNaN(val) || Number(val) <= 0) {
      val = hojaConfig.getRange("A2").getValue();
    }
    if (!isNaN(val) && Number(val) > 0) {
      valorOroDia = Number(val);
    }
  } catch (e) {}

  return { status: "success", valor_oro_dia: valorOroDia };
}

function actualizarValorOroDia_(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var valorNuevo = Number(data.valor_oro_dia);
  if (isNaN(valorNuevo) || valorNuevo <= 0) {
    return { status: "error", message: "El valor del oro ingresado no es válido." };
  }

  var hojaConfig = ss.getSheetByName("ORO_DIA") || ss.getSheetByName("CONFIG") || ss.getSheetByName("ACTUALIZACION_ORO");
  if (!hojaConfig) {
    hojaConfig = ss.insertSheet("ORO_DIA");
    hojaConfig.getRange("A1").setValue("VALOR_ORO_DIA");
  }

  hojaConfig.getRange("B1").setValue(valorNuevo);
  registrarEnEspejo_("ACTUALIZAR_ORO", "GENERAL", "Actualización del valor del oro del día a $" + valorNuevo.toLocaleString(), data.usuario || "ADMIN");
  return { status: "success", message: "¡Valor del oro del día actualizado con éxito a $" + valorNuevo.toLocaleString() + "!" };
}

function registrarEntradaSalida_(nombreHoja, data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hoja = ss.getSheetByName(nombreHoja);
  
  if (!hoja) {
    hoja = ss.insertSheet(nombreHoja);
    if (nombreHoja === "SALIDAS") {
      hoja.appendRow(["ID_Salida", "Fecha", "SKU", "Cantidad", "Motivo", "Usuario", "Observaciones"]);
    } else {
      hoja.appendRow(["ID_Movimiento", "Fecha", "SKU", "Cantidad", "Motivo", "Usuario", "Observaciones"]);
    }
  }

  var idMov = (nombreHoja === "SALIDAS" ? "SAL-" : "MOV-") + Utilities.getUuid().substring(0, 8).toUpperCase();
  var fecha = Utilities.formatDate(new Date(), "America/Bogota", "yyyy-MM-dd HH:mm:ss");
  var identificadorBuscado = String(data.sku || "").trim();
  
  if (nombreHoja === "SALIDAS") {
    hoja.appendRow([
      idMov,
      fecha,
      identificadorBuscado,
      data.cantidad || 1,
      data.motivo || "VENTA",
      data.usuario || "ADMIN",
      data.observaciones || ""
    ]);

    var hojaProductos = ss.getSheetByName("PRODUCTOS");
    if (hojaProductos && hojaProductos.getLastRow() > 1) {
      var prodRango = hojaProductos.getDataRange().getValues();
      var prodHeaders = prodRango[0];
      var skuIdxProd = prodHeaders.indexOf("SKU");
      var barcodeIdxProd = prodHeaders.indexOf("Codigo_Barra");
      var estadoIdxProd = prodHeaders.indexOf("Estado");

      if (estadoIdxProd !== -1) {
        for (var p = 1; p < prodRango.length; p++) {
          var rowSku = skuIdxProd !== -1 ? String(prodRango[p][skuIdxProd]).trim() : "";
          var rowBarcode = barcodeIdxProd !== -1 ? String(prodRango[p][barcodeIdxProd]).trim() : "";
          
          if (rowSku.toUpperCase() === identificadorBuscado.toUpperCase() || rowBarcode.toUpperCase() === identificadorBuscado.toUpperCase()) {
            hojaProductos.getRange(p + 1, estadoIdxProd + 1).setValue("VENDIDO");
            break;
          }
        }
      }
    }

    registrarEnEspejo_("SALIDA_INVENTARIO", identificadorBuscado, "Salida registrada por motivo: " + (data.motivo || "VENTA") + " | Obs: " + (data.observaciones || ""), data.usuario || "ADMIN");

  } else {
    hoja.appendRow([
      idMov,
      fecha,
      identificadorBuscado,
      data.cantidad || 1,
      data.motivo || "GENERAL",
      data.usuario || "ADMIN",
      data.observaciones || ""
    ]);

    var hojaProductos = ss.getSheetByName("PRODUCTOS");
    if (hojaProductos && hojaProductos.getLastRow() > 1) {
      var prodRango = hojaProductos.getDataRange().getValues();
      var prodHeaders = prodRango[0];
      var skuIdxProd = prodHeaders.indexOf("SKU");
      var barcodeIdxProd = prodHeaders.indexOf("Codigo_Barra");
      var estadoIdxProd = prodHeaders.indexOf("Estado");

      if (estadoIdxProd !== -1) {
        for (var p = 1; p < prodRango.length; p++) {
          var rowSku = skuIdxProd !== -1 ? String(prodRango[p][skuIdxProd]).trim() : "";
          var rowBarcode = barcodeIdxProd !== -1 ? String(prodRango[p][barcodeIdxProd]).trim() : "";
          
          if (rowSku.toUpperCase() === identificadorBuscado.toUpperCase() || rowBarcode.toUpperCase() === identificadorBuscado.toUpperCase()) {
            hojaProductos.getRange(p + 1, estadoIdxProd + 1).setValue("DISPONIBLE");
            break;
          }
        }
      }
    }

    registrarEnEspejo_("ENTRADA_INVENTARIO", identificadorBuscado, "Entrada/Devolución registrada por motivo: " + (data.motivo || "GENERAL") + " | Obs: " + (data.observaciones || ""), data.usuario || "ADMIN");
  }

  return { status: "success", message: "¡Registro de " + nombreHoja.toLowerCase() + " guardado con éxito y auditoría espejo actualizada!" };
}

function reversarSalida_(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hojaSalidas = ss.getSheetByName("SALIDAS");
  if (!hojaSalidas) return { status: "error", message: "Hoja SALIDAS no encontrada." };

  var filas = hojaSalidas.getDataRange().getValues();
  var headers = filas[0];
  var idIdx = headers.indexOf("ID_Salida");
  if (idIdx === -1) idIdx = 0;
  var skuIdx = headers.indexOf("SKU");

  var targetId = String(data.id_salida || "").trim();

  for (var r = 1; r < filas.length; r++) {
    var rowId = String(filas[r][idIdx] || "").trim();
    if (rowId === targetId) {
      var skuAfectado = skuIdx !== -1 ? String(filas[r][skuIdx] || "").trim() : "";
      
      // Eliminar el registro de salidas
      hojaSalidas.deleteRow(r + 1);

      // Restaurar el estado del producto a DISPONIBLE en PRODUCTOS
      var hojaProductos = ss.getSheetByName("PRODUCTOS");
      if (hojaProductos && hojaProductos.getLastRow() > 1) {
        var prodRango = hojaProductos.getDataRange().getValues();
        var prodHeaders = prodRango[0];
        var pSkuIdx = prodHeaders.indexOf("SKU");
        var pBarIdx = prodHeaders.indexOf("Codigo_Barra");
        var pEstadoIdx = prodHeaders.indexOf("Estado");

        if (pEstadoIdx !== -1) {
          for (var p = 1; p < prodRango.length; p++) {
            var rSku = pSkuIdx !== -1 ? String(prodRango[p][pSkuIdx]).trim() : "";
            var rBar = pBarIdx !== -1 ? String(prodRango[p][pBarIdx]).trim() : "";

            if (rSku.toUpperCase() === skuAfectado.toUpperCase() || rBar.toUpperCase() === skuAfectado.toUpperCase()) {
              hojaProductos.getRange(p + 1, pEstadoIdx + 1).setValue("DISPONIBLE");
              break;
            }
          }
        }
      }

      registrarEnEspejo_("REVERSAR_SALIDA", skuAfectado, "Se reversó/anuló la salida ID: " + targetId + " y el producto volvió a DISPONIBLE", data.usuario || "ADMIN");
      return { status: "success", message: "¡Salida reversada con éxito! El producto ha vuelto a estar disponible." };
    }
  }
  return { status: "error", message: "No se encontró el registro de salida para reversar." };
}

function crearOActualizarProducto_(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hoja = ss.getSheetByName("PRODUCTOS");
  var productos = obtenerDatosHoja_("PRODUCTOS");

  var productoExistente = null;
  for (var i = 0; i < productos.length; i++) {
    var p = productos[i];
    var mismaDescripcion = p.Nombre && p.Nombre.toLowerCase() === data.nombre.toLowerCase() && p.ID_Categoria === data.categoria;
    var mismoCodigoBarra = data.codigo_barra && p.Codigo_Barra && p.Codigo_Barra === data.codigo_barra;
    
    if (mismoCodigoBarra || mismaDescripcion) {
      productoExistente = p;
      break;
    }
  }

  if (productoExistente && !data.forzar_creacion) {
    return { 
      status: "existe", 
      message: "¡El producto '" + productoExistente.Nombre + "' ya se encuentra registrado con el SKU " + productoExistente.SKU + "!",
      producto: productoExistente
    };
  }

  var catPrefijo = "JO";
  if (data.categoria) {
    catPrefijo = data.categoria.substring(0, 2).toUpperCase();
  }
  
  var consecutivo = productos.length + 1;
  var skuGenerado = data.sku || (catPrefijo + String(consecutivo).padStart(3, '0'));
  var codigoBarraGenerado = data.codigo_barra || ("770" + Math.floor(100000000 + Math.random() * 900000000));
  var idProducto = "PRD-" + Utilities.getUuid().substring(0, 8).toUpperCase();
  var fechaCreacion = Utilities.formatDate(new Date(), "America/Bogota", "yyyy-MM-dd");

  var valorPiedra = data.valor_piedra !== "" && !isNaN(data.valor_piedra) ? Number(data.valor_piedra) : 0;
  var valorOro = data.valor_oro !== "" && !isNaN(data.valor_oro) ? Number(data.valor_oro) : 0;
  var valorCompraTotal = valorPiedra + valorOro;
  
  var porcentajeVenta = data.porcentaje_venta !== "" && !isNaN(data.porcentaje_venta) ? Number(data.porcentaje_venta) : 100;
  var colorSeleccionado = data.color || data.ID_Subcategoria || "";

  var headers = hoja.getRange(1, 1, 1, hoja.getLastColumn()).getValues()[0];
  var nuevaFila = new Array(headers.length);

  headers.forEach(function(h, index) {
    if (h === "ID_Producto") nuevaFila[index] = idProducto;
    else if (h === "SKU") nuevaFila[index] = skuGenerado;
    else if (h === "Ref") nuevaFila[index] = skuGenerado;
    else if (h === "Codigo_Barra") nuevaFila[index] = codigoBarraGenerado;
    else if (h === "Nombre") nuevaFila[index] = data.nombre;
    else if (h === "Descripcion") nuevaFila[index] = "Piedra: " + (data.piedra || "N/A") + " | Color: " + colorSeleccionado;
    else if (h === "ID_Categoria") nuevaFila[index] = data.categoria || "";
    else if (h === "Material" || h === "ID_Proveedor" || h === "Material_Oro") nuevaFila[index] = data.material || data.Material_Oro || "";
    else if (h === "Color" || h === "ID_Subcategoria") nuevaFila[index] = colorSeleccionado;
    else if (h === "ID_Ubicacion") nuevaFila[index] = data.ubicacion || "";
    else if (h === "Stock_Min") nuevaFila[index] = 2;
    else if (h === "Stock_Max") nuevaFila[index] = 50;
    else if (h === "Estado") nuevaFila[index] = "DISPONIBLE";
    else if (h === "Fecha_Creacion") nuevaFila[index] = fechaCreacion;
    else if (h === "Peso") nuevaFila[index] = data.peso !== "" ? String(data.peso).replace(",", ".") : "";
    else if (h === "Valor_Compra") nuevaFila[index] = valorCompraTotal;
    else if (h === "Porcentaje_Venta") nuevaFila[index] = porcentajeVenta;
    else if (h === "Tiene_Descuento") nuevaFila[index] = data.tiene_descuento !== "" ? Number(data.tiene_descuento) : 0;
    else if (h === "Valor_Piedra") nuevaFila[index] = valorPiedra;
    else if (h === "Valor_Oro") nuevaFila[index] = valorOro;
    else if (h === "Foto") nuevaFila[index] = data.foto || "";
  });

  var filaNum = hoja.getLastRow() + 1;
  hoja.appendRow(nuevaFila);

  var fechaCreacionIdx = headers.indexOf("Fecha_Creacion");
  if (fechaCreacionIdx !== -1) {
    hoja.getRange(filaNum, fechaCreacionIdx + 1).setNumberFormat("@");
    hoja.getRange(filaNum, fechaCreacionIdx + 1).setValue(fechaCreacion);
  }

  var pesoIdx = headers.indexOf("Peso");
  if (pesoIdx !== -1 && data.peso !== "") {
    hoja.getRange(filaNum, pesoIdx + 1).setNumberFormat("@");
    hoja.getRange(filaNum, pesoIdx + 1).setValue(String(data.peso).replace(",", "."));
  }

  registrarEnEspejo_("CREAR_PRODUCTO", skuGenerado, "Creación de joya: " + data.nombre + " [Categoría: " + data.categoria + "]", data.usuario || "ADMIN");
  return { status: "success", message: "¡Producto registrado con éxito! SKU Asignado: " + skuGenerado, id_producto: idProducto };
}

function editarProducto_(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hoja = ss.getSheetByName("PRODUCTOS");
  var filas = hoja.getDataRange().getValues();
  var headers = filas[0];
  
  var idIdx = headers.indexOf("ID_Producto");
  var skuIdx = headers.indexOf("SKU");
  
  var targetId = String(data.id_producto || "").trim();
  var targetSku = String(data.sku || "").trim();

  for (var r = 1; r < filas.length; r++) {
    var currentRowId = idIdx !== -1 ? String(filas[r][idIdx] || "").trim() : "";
    var currentRowSku = skuIdx !== -1 ? String(filas[r][skuIdx] || "").trim() : "";
    
    var coincide = (targetId && currentRowId === targetId) || (targetSku && currentRowSku === targetSku);

    if (coincide) {
      var filaNum = r + 1;
      
      if (headers.indexOf("Nombre") !== -1 && data.nombre !== undefined) hoja.getRange(filaNum, headers.indexOf("Nombre") + 1).setValue(data.nombre);
      if (headers.indexOf("ID_Categoria") !== -1 && data.categoria !== undefined) hoja.getRange(filaNum, headers.indexOf("ID_Categoria") + 1).setValue(data.categoria);
      if ((headers.indexOf("Material") !== -1 || headers.indexOf("ID_Proveedor") !== -1) && data.material !== undefined) {
        var idxMat = headers.indexOf("Material") !== -1 ? headers.indexOf("Material") : headers.indexOf("ID_Proveedor");
        hoja.getRange(filaNum, idxMat + 1).setValue(data.material);
      }
      
      if ((headers.indexOf("Color") !== -1 || headers.indexOf("ID_Subcategoria") !== -1) && data.color !== undefined) {
        var idxCol = headers.indexOf("Color") !== -1 ? headers.indexOf("Color") : headers.indexOf("ID_Subcategoria");
        hoja.getRange(filaNum, idxCol + 1).setValue(data.color);
      }

      if (headers.indexOf("ID_Ubicacion") !== -1 && data.ubicacion !== undefined) hoja.getRange(filaNum, headers.indexOf("ID_Ubicacion") + 1).setValue(data.ubicacion);
      
      var vPiedra = data.valor_piedra !== undefined && data.valor_piedra !== "" ? Number(data.valor_piedra) : null;
      var vOro = data.valor_oro !== undefined && data.valor_oro !== "" ? Number(data.valor_oro) : null;
      
      if (vPiedra !== null && headers.indexOf("Valor_Piedra") !== -1) hoja.getRange(filaNum, headers.indexOf("Valor_Piedra") + 1).setValue(vPiedra);
      if (vOro !== null && headers.indexOf("Valor_Oro") !== -1) hoja.getRange(filaNum, headers.indexOf("Valor_Oro") + 1).setValue(vOro);
      
      if (vPiedra !== null || vOro !== null) {
        var actualPiedra = headers.indexOf("Valor_Piedra") !== -1 ? Number(hoja.getRange(filaNum, headers.indexOf("Valor_Piedra") + 1).getValue()) || 0 : 0;
        var actualOro = headers.indexOf("Valor_Oro") !== -1 ? Number(hoja.getRange(filaNum, headers.indexOf("Valor_Oro") + 1).getValue()) || 0 : 0;
        if (headers.indexOf("Valor_Compra") !== -1) {
          hoja.getRange(filaNum, headers.indexOf("Valor_Compra") + 1).setValue(actualPiedra + actualOro);
        }
      }

      if (headers.indexOf("Porcentaje_Venta") !== -1 && data.porcentaje_venta !== undefined) hoja.getRange(filaNum, headers.indexOf("Porcentaje_Venta") + 1).setValue(data.porcentaje_venta !== "" ? Number(data.porcentaje_venta) : 100);
      if (headers.indexOf("Tiene_Descuento") !== -1 && data.tiene_descuento !== undefined) hoja.getRange(filaNum, headers.indexOf("Tiene_Descuento") + 1).setValue(data.tiene_descuento !== "" ? Number(data.tiene_descuento) : 0);
      
      if (headers.indexOf("Peso") !== -1 && data.peso !== undefined) {
        var celdaPeso = hoja.getRange(filaNum, headers.indexOf("Peso") + 1);
        celdaPeso.setNumberFormat("@");
        celdaPeso.setValue(data.peso !== "" ? String(data.peso).replace(",", ".") : "");
      }
      
      if (headers.indexOf("Foto") !== -1 && data.foto) {
        hoja.getRange(filaNum, headers.indexOf("Foto") + 1).setValue(data.foto);
      }

      registrarEnEspejo_("EDITAR_PRODUCTO", targetSku || currentRowSku, "Modificación de producto: " + (data.nombre || "Sin nombre"), data.usuario || "ADMIN");
      return { status: "success", message: "¡Producto actualizado correctamente!" };
    }
  }
  return { status: "error", message: "Producto no encontrado para editar." };
}

function eliminarMasivoProductos_(data) {

      eliminadosCount++;

function obtenerDatosHoja_(nombreHoja) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hoja = ss.getSheetByName(nombreHoja);
  if (!hoja || hoja.getLastRow() <= 1) return [];
  
  var rango = hoja.getDataRange().getValues();
  var headers = rango[0];
  var rows = rango.slice(1);
  
  return rows.map(function(row) {
    var obj = {};
    headers.forEach(function(header, i) {
      var val = row[i];
      if (val instanceof Date) {
        obj[header] = Utilities.formatDate(val, "America/Bogota", "yyyy-MM-dd");
      } else {
        if (header.toLowerCase() === "peso" && val !== null && val !== undefined) {
          obj[header] = String(val).replace(",", ".");
        } else {
          obj[header] = val !== undefined ? val : "";
        }
      }
    });
    if (!obj["Color"] && obj["ID_Subcategoria"]) {
      obj["Color"] = obj["ID_Subcategoria"];
    }
    return obj;
  });
}

function obtenerDatosConDetalleProducto_(nombreHoja) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var movimientos = obtenerDatosHoja_(nombreHoja);
  var productos = obtenerDatosHoja_("PRODUCTOS");

  var mapaProductos = {};
  for (var i = 0; i < productos.length; i++) {
    var p = productos[i];
    var skuKey = String(p.SKU || p.Ref || "").trim().toUpperCase();
    var barKey = String(p.Codigo_Barra || "").trim().toUpperCase();
    
    var infoProd = {
      nombre: p.Nombre || "",
      codigo_barra: p.Codigo_Barra || "",
      estado: p.Estado || p.estado || "DISPONIBLE"
    };

    if (skuKey) mapaProductos[skuKey] = infoProd;
    if (barKey) mapaProductos[barKey] = infoProd;
  }

  for (var j = 0; j < movimientos.length; j++) {
    var m = movimientos[j];
    var skuMov = String(m.SKU || m.sku || "").trim().toUpperCase();
    
    var datosProd = mapaProductos[skuMov] || { nombre: "Producto no encontrado", codigo_barra: skuMov, estado: "DESCONOCIDO" };
    
    m.Nombre_Producto = datosProd.nombre;
    m.Codigo_Barra = datosProd.codigo_barra;
    m.Estado_Actual = datosProd.estado;
  }

  return movimientos;
}

// FUNCIÓN ESPEJO DE AUDITORÍA: Guarda Fecha, Hora, Usuario, Acción y Detalle en la pestaña HISTORIAL_CAMBIOS
function registrarEnEspejo_(accion, sku, detalle, usuario) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hojaEspejo = ss.getSheetByName("HISTORIAL_CAMBIOS");

    hojaEspejo = ss.insertSheet("HISTORIAL_CAMBIOS");
    hojaEspejo.appendRow(["ID_Bitacora", "Fecha_Hora", "Accion", "SKU", "Detalle_Modificacion", "Usuario"]);
  }

  var idBitacora = "BIT-" + Utilities.getUuid().substring(0, 8).toUpperCase();
  var fechaHora = Utilities.formatDate(new Date(), "America/Bogota", "yyyy-MM-dd HH:mm:ss");
  
  hojaEspejo.appendRow([
    idBitacora,
    fechaHora,
    String(accion || "").toUpperCase(),
